### firda-server

```powershell
param(
  [string]$Serial = "",
  [string]$Version = "",
  [switch]$Restart
)

$ErrorActionPreference = "Stop"

function Get-AndroidSdk {
  if ($env:ANDROID_HOME) { return $env:ANDROID_HOME }
  if ($env:ANDROID_SDK_ROOT) { return $env:ANDROID_SDK_ROOT }
  return Join-Path $env:LOCALAPPDATA "Android\Sdk"
}

function Get-FridaExe {
  $cmd = Get-Command frida.exe -ErrorAction SilentlyContinue
  if ($cmd) { return $cmd.Source }

  python -m pip install --user frida-tools | Out-Null

  $python = (Get-Command python).Source
  $scripts = Join-Path (Split-Path $python -Parent) "Scripts"
  $candidate = Join-Path $scripts "frida.exe"
  if (Test-Path $candidate) { return $candidate }

  $userCandidate = Join-Path $env:APPDATA "Python\Python312\Scripts\frida.exe"
  if (Test-Path $userCandidate) { return $userCandidate }

  throw "frida.exe not found after installing frida-tools"
}

function Invoke-Adb {
  param([string[]]$AdbArgs)
  $fullArgs = @()
  if ($script:Serial) { $fullArgs += @("-s", $script:Serial) }
  $fullArgs += $AdbArgs
  & $script:adb @fullArgs
}

function Select-EmulatorSerial {
  if ($script:Serial) { return }

  $devices = & $script:adb devices
  $emulators = @()
  foreach ($line in $devices) {
    if ($line -match '^(emulator-\d+)\s+device\b') {
      $emulators += $matches[1]
    }
  }

  if ($emulators.Count -eq 0) {
    throw "No online emulator device found. Refusing to use non-emulator devices."
  }
  if ($emulators.Count -gt 1) {
    throw "Multiple online emulators found: $($emulators -join ', '). Pass -Serial explicitly."
  }

  $script:Serial = $emulators[0]
  Write-Host "Using emulator serial: $script:Serial"
}

$sdk = Get-AndroidSdk
$script:adb = Join-Path $sdk "platform-tools\adb.exe"
if (!(Test-Path $script:adb)) { throw "adb not found: $script:adb" }
if ($Serial) { $script:Serial = $Serial }

$frida = Get-FridaExe
if (!$Version) {
  $Version = (& $frida --version).Trim()
}

Select-EmulatorSerial
Invoke-Adb -AdbArgs @("wait-for-device")
$abi = (Invoke-Adb -AdbArgs @("shell", "getprop", "ro.product.cpu.abi")).Trim()
$arch = switch ($abi) {
  "x86_64" { "android-x86_64" }
  "x86" { "android-x86" }
  "arm64-v8a" { "android-arm64" }
  default { "android-arm" }
}

$toolsDir = Join-Path (Get-Location) "tools\frida"
New-Item -ItemType Directory -Force -Path $toolsDir | Out-Null

$serverName = "frida-server-$Version-$arch"
$serverPath = Join-Path $toolsDir $serverName
$xzPath = "$serverPath.xz"

if (!(Test-Path $serverPath)) {
  $url = "https://github.com/frida/frida/releases/download/$Version/$serverName.xz"
  Write-Host "Downloading $url"
  Invoke-WebRequest -Uri $url -OutFile $xzPath

  Write-Host "Decompressing $xzPath"
  python -c "import lzma, pathlib; src=pathlib.Path(r'$xzPath'); dst=pathlib.Path(r'$serverPath'); dst.write_bytes(lzma.decompress(src.read_bytes()))"
}

Write-Host "Restarting adb as root. Play Store images usually fail here; use a Google APIs/AOSP image."
Invoke-Adb -AdbArgs @("root")
Start-Sleep -Seconds 2
Invoke-Adb -AdbArgs @("wait-for-device")

if ($Restart) {
  Invoke-Adb -AdbArgs @("shell", "pkill", "-f", "frida-server") | Out-Null
}

Invoke-Adb -AdbArgs @("push", $serverPath, "/data/local/tmp/frida-server")
Invoke-Adb -AdbArgs @("shell", "chmod", "755", "/data/local/tmp/frida-server")
Invoke-Adb -AdbArgs @("shell", "pkill", "-f", "frida-server") | Out-Null

# 安卓里面的命令
# nohup                               后台运行，不随 shell 退出而停止
# /data/local/tmp/frida-server        要启动的程序
# > /data/local/tmp/frida-server.log  把标准输出写到日志
# 2>&1                                把错误输出也写到同一个日志
# &                                   放到后台运行
Invoke-Adb -AdbArgs @("shell", "nohup /data/local/tmp/frida-server >/data/local/tmp/frida-server.log 2>&1 &")
Start-Sleep -Seconds 2

$fridaPs = $frida -replace "frida\.exe$", "frida-ps.exe"
if (!(Test-Path $fridaPs)) { $fridaPs = "frida-ps" }
& $fridaPs -D $script:Serial

Write-Host "frida-server is running on device ABI $abi with Frida $Version"

```

### 把 frida-server 放到雷电模拟器中


``` powershell
powershell -ExecutionPolicy Bypass -File .\scripts\frida-server.ps1 `
  -Serial emulator-5554 `
  -Restart
  
# Windows 默认可能不让你直接运行 `.ps1` 文件
# 所以下面的参数表示：这一次运行脚本时，临时绕过 Windows 的 PowerShell 脚本执行限制。
-ExecutionPolicy Bypass  

# 意思是：运行这个脚本文件。
-File .\scripts\frida-server.ps1

# 我们脚本自己的参数 
-Serial emulator-5554

# 我们脚本自己的参数
-Restart


# 最后那个反引号 ` 是 PowerShell 的换行符。等价于一行写法：
powershell -ExecutionPolicy Bypass -File .\scripts\frida-server.ps1 -Serial emulator-5554 -Restart

# ⭐ 重要命令, 上面的 .ps1 都是对这个的包装
& $frida -D emulator-5554 -f com.whatsapp -l D:\learn\android-mock\1\hooks\minimal-api-tracer.js

# -l D:\learn\android-mock\1\hooks\minimal-api-tracer.js
# 加载这个 JS hook 文件
```


### Frida JS Hook

核心就这几个：

``` js
Java.perform(function () {
  // 等 Android Java 环境准备好
});

const Debug = Java.use('android.os.Debug');
// 拿到 Android Java 类

Debug.isDebuggerConnected.implementation = function () {
  // 替换这个 Java 方法的实现
};

const result = Debug.isDebuggerConnected.call(this);
// 调原始方法，拿真实返回值



// Java.use(...)
// 拿到 Android 里的某个 Java 类。

// Debug.isDebuggerConnected
// 拿到这个类里的某个方法。

// implementation = function () {}
// 把这个方法替换成我们自己的函数。

// call(this, ...)
// 在我们自己的函数里，再调用原始方法，拿真实结果。
```
