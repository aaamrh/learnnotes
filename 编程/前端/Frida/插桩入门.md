# 写个 Hook

``` js

Java.perform(function () {
  var F8k = Java.use("X.F8k");

  var A0m = F8k.A0m.overload("java.lang.String", "java.lang.String");

  A0m.implementation = function (a, b) {
    console.log("");
    console.log("[+] X.F8k.A0m called");
    console.log("arg0 = " + a);
    console.log("arg1 = " + b);

    var ret = A0m.call(this, a, b);

    console.log("ret  = " + ret);
    console.log("[+] X.F8k.A0m finished");
    console.log("==");

    return ret;
  };

  console.log("[*] hook installed: X.F8k.A0m(String, String)");
});

```

# demo

**. hook 构造函数 $init**

例子：hook `java.io.File(String)`，看 App 创建了哪些文件路径。

```
Java.perform(function () {
  var File = Java.use("java.io.File");

  File.$init.overload("java.lang.String").implementation = function (path) {
    console.log("[File.$init] path = " + path);
    return this.$init(path);
  };
});
```

构造函数固定写法：

```
Class.$init.overload(...).implementation = function (...) {
  return this.$init(...);
};
```

**2. hook 静态方法**

例子：hook `android.util.Base64.encodeToString(byte[], int)`。

```
Java.perform(function () {
  var Base64 = Java.use("android.util.Base64");

  Base64.encodeToString.overload("[B", "int").implementation = function (bytes, flags) {
    console.log("[Base64.encodeToString]");
    console.log("flags = " + flags);
    console.log("bytes length = " + bytes.length);

    var ret = Base64.encodeToString(bytes, flags);

    console.log("ret = " + ret);
    return ret;
  };
});
```

静态方法用：

```
ClassName.method(...)
```

实例方法用：

```
this.method(...)
```

**3. hook 所有 overload**

如果你不知道参数类型，就把所有重载都 hook：

```
Java.perform(function () {
  var Log = Java.use("android.util.Log");

  Log.d.overloads.forEach(function ( overload ) {
    overload.implementation = function () {
      console.log("[Log.d overload]");
      for (var i = 0; i < arguments.length; i++) {
        console.log("arg" + i + " = " + arguments[i]);
      }

      return overload.apply(this, arguments);
    };
  });
});
```

通用模板：

```
Class.method.overloads.forEach(function (m) {
  m.implementation = function () {
    console.log(arguments);
    return m.apply(this, arguments);
  };
});
```

**4. 修改参数**

例子：hook 你现在的 `X.F8k.A0m(String, String)`，把第二个参数改掉。

```
Java.perform(function () {
  var F8k = Java.use("X.F8k");
  var A0m = F8k.A0m.overload("java.lang.String", "java.lang.String");

  A0m.implementation = function (a, b) {
    console.log("[A0m before]");
    console.log("a = " + a);
    console.log("b = " + b);

    var newB = "1234567890";

    console.log("[A0m modified]");
    console.log("newB = " + newB);

    return A0m.call(this, a, newB);
  };
});
```

核心就是：

```
return 原函数.call(this, 原参数, 新参数);
```

**5. 修改返回值**

例子：hook `String.contains()`，强行返回 `false`。

```
Java.perform(function () {
  var StringCls = Java.use("java.lang.String");

  StringCls.contains.implementation = function (s) {
    var ret = this.contains(s);

    console.log("[String.contains]");
    console.log("this = " + this);
    console.log("search = " + s);
    console.log("origin ret = " + ret);

    return false;
  };
});
```

真实调试时不要乱改返回值，先打印，确认逻辑后再改。

**6. 打印对象字段**

假设 JADX 里看到：

```
obj.A00
obj.A01
```

Frida 里这样读：

```
Java.perform(function () {
  var F8k = Java.use("X.F8k");
  var A0m = F8k.A0m.overload("java.lang.String", "java.lang.String");

  A0m.implementation = function (a, b) {
    console.log("[this object] " + this);

    try {
      console.log("this.A00 = " + this.A00.value);
    } catch (e) {
      console.log("read A00 failed: " + e);
    }

    try {
      console.log("this.A01 = " + this.A01.value);
    } catch (e) {
      console.log("read A01 failed: " + e);
    }

    return A0m.call(this, a, b);
  };
});
```

字段一般用：

```
this.字段名.value
```

**7. Java.choose 查内存对象**

例子：找内存里已有的 `X.F8k` 实例。

```
Java.perform(function () {
  Java.choose("X.F8k", {
    onMatch: function (obj) {
      console.log("[Java.choose] found X.F8k instance: " + obj);

      try {
        console.log("A00 = " + obj.A00.value);
      } catch (e) {
        console.log("read field failed: " + e);
      }
    },
    onComplete: function () {
      console.log("[Java.choose] done");
    }
  });
});
```

用途：

```
不用等函数触发，直接从内存里找某个类的对象。
```

**8. Interceptor.attach hook native so**

例子：hook libc 的 `open`，看 native 层打开了哪些文件。

```
var openPtr = Module.findExportByName("libc.so", "open");

Interceptor.attach(openPtr, {
  onEnter: function (args) {
    this.path = args[0].readCString();
    console.log("[native open] path = " + this.path);
  },
  onLeave: function (retval) {
    console.log("[native open] fd = " + retval);
  }
});
```

native hook 常见格式：

```
var addr = Module.findExportByName("libxxx.so", "函数名");

Interceptor.attach(addr, {
  onEnter: function (args) {
    console.log(args[0]);
  },
  onLeave: function (retval) {
    console.log(retval);
  }
});
```

**9. send() 发回 Python**

JS 脚本：

```
Java.perform(function () {
  var F8k = Java.use("X.F8k");
  var A0m = F8k.A0m.overload("java.lang.String", "java.lang.String");

  A0m.implementation = function (a, b) {
    send({
      event: "A0m_called",
      arg0: a ? a.toString() : null,
      arg1: b ? b.toString() : null
    });

    return A0m.call(this, a, b);
  };
});
```

Python 接收：

```
import frida
import sys

device = frida.get_usb_device()
session = device.attach("WhatsApp")

with open(r"C:\Users\aaamr\Desktop\hook-whatsapp.js", "r", encoding="utf-8") as f:
    script = session.create_script(f.read())

def on_message(message, data):
    print(message)

script.on("message", on_message)
script.load()

sys.stdin.read()
```

但你现在先用 `console.log` 就够了。`send()` 更适合后面把日志保存成文件、JSON、CSV。

**最常用顺序**

你实际分析 App 时，按这个顺序用：

```
1. hook 指定函数参数
2. 打印调用栈
3. hook 所有 overload
4. 打印 this 字段
5. Java.choose 找对象
6. 必要时修改参数/返回值
7. Java 层看不到时，再 hook native so
```
