# 合并处理同一接口的并发请求
> 场景: 
  接口是支持 [id1,id2,id3]这种形式批量获取数据的。
  每个 webview 从 render 调用方法获取数据，如果 webview 多了可能会有大的并发请求，所以可能等 50 ms, 收集所有请求参数，然后发一个请求就好。
```javascript
export function createDebouncedBatcher<TId extends string, TResult, TItem = TResult>(
  fetcher: (ids: TId[]) => Promise<TResult>,
  options: DebouncedBatcherOptions<TId, TResult, TItem> = {}
): DebouncedBatcher<TId, TResult, TItem> {
  const defer = options.defer ?? 50;
  const pickById = options.pickById;

  let timer: ReturnType<typeof setTimeout> | null = null;
  let pendingIds = new Set<TId>();
  let waitersById = new Map<TId, Array<DebouncedBatcherDeferred<TItem>>>();
  let flushing = false;

  const cancelTimer = () => {
    if (timer) {
      clearTimeout(timer);
      timer = null;
    }
  };

  const scheduleFlush = (flushFn: () => void) => {
    if (timer) return;
    timer = setTimeout(() => {
      timer = null;
      flushFn();
    }, defer);
  };

  const flush = async (): Promise<void> => {
    if (flushing) return;
    if (pendingIds.size === 0) return;

    flushing = true;

    // 快照并清空（新 enqueue 进入下一批）
    const ids = Array.from(pendingIds);
    const localWaiters = waitersById;

    pendingIds = new Set<TId>();
    waitersById = new Map<TId, Array<DebouncedBatcherDeferred<TItem>>>();

    try {
      const result = await fetcher(ids);

      // 将同一批次 result 分发给每个 id 的所有 waiter
      for (const id of ids) {
        const waiters = localWaiters.get(id);
        if (!waiters || waiters.length === 0) continue;
        const picked = pickById ? pickById(result, id) : (result as unknown as TItem);
        waiters.forEach(({ resolve }) => resolve(picked));
      }
    } catch (error) {
      for (const id of ids) {
        const waiters = localWaiters.get(id);
        if (!waiters || waiters.length === 0) continue;
        waiters.forEach(({ reject }) => reject(error));
      }
    } finally {
      flushing = false;

      // 如果 flush 过程中有新 pending，尽快安排下一轮（不再等待 defer）
      if (pendingIds.size > 0 && !timer) {
        timer = setTimeout(() => {
          timer = null;
          void flush();
        }, 0);
      }
    }
  };

  const enqueue = (id: TId): Promise<TItem> => {
    pendingIds.add(id);

    const promise = new Promise<TItem>((resolve, reject) => {
      const list = waitersById.get(id) ?? [];
      list.push({ resolve, reject });
      waitersById.set(id, list);
    });

    scheduleFlush(() => {
      void flush();
    });

    return promise;
  };

  const flushNow = async (): Promise<void> => {
    cancelTimer();
    await flush();
  };

  return {
    enqueue,
    flushNow,
    cancelTimer,
    getPendingCount: () => pendingIds.size,
    isFlushing: () => flushing,
  };
}

```

# Promise.withResolves

这个很巧妙，和无感刷新token一样，保存了 resolve。在需要的时候调用，尤其在判断什么东西是否准备好了的场景很好用。

``` javascript
function withResolves(){
	let done, fail;
	const run = new Promise((resolve, reject) => {
		done = resolve;
		fail = reject;
	})
	return {done, fail, run};
}
```

# 惰性单例

**场景：** 你有一个很重的 SDK（比如地图、IM、支付），或者一个很贵的计算。你不想页面一加载就初始化，而想等到**第一次有人调用它**时再初始化，且只初始化一次。 **普通做法：** 全局变量 `let instance = null`，每次用的时候 `if (!instance) init()`。有点啰嗦。 **高级技巧：** **重写自身函数 (Function Overwriting)。**

```JavaScript
let getHeavySDK = async () => {
    console.log('🚧 第一次调用：正在初始化...');
    
    // 1. 执行初始化
    const sdk = await import('./heavy-sdk'); 
    const instance = sdk.createInstance();

    // 2. 【核心技巧】重写函数本身！
    // 下次再调用 getHeavySDK，直接执行下面这个函数，不再走上面的逻辑
    getHeavySDK = async () => {
        console.log('🚀 后续调用：直接返回实例');
        return instance;
    };

    return instance;
};

// --- 使用场景 ---
await getHeavySDK(); // 打印：🚧 第一次调用...
await getHeavySDK(); // 打印：🚀 后续调用... (连 if 判断都省了)
```

**原理：** 这是一个极其纯粹的性能优化，利用 JS 函数是一等公民的特性，在运行时动态修改函数的定义。

```html
 <script>
      class AsyncSeriesWaterfallHook {
        constructor(args = []) {
          this._args = args; // 示例: ["source", "config"]
          this.taps = [];
        }

        // 注册异步回调插件
        tapAsync(name, fn) {
          this.taps.push(fn);
        }

        compile() {
          const params = this._args.join(", "); // "source, config"
          const firstArg = this._args[0]; // "source"

          // 动态生成执行逻辑
          // 核心思想：手动构建一个递归函数 next(i, currentSource)
          let code = `
          var taps = _x;
          var index = 0;

          function next(err, newValue) {
            if (err) return finalCallback(err);
            if (newValue !== undefined) {
              ${firstArg} = newValue; // 更新流水线的值
            }

            if (index >= taps.length) {
              return finalCallback(null, ${firstArg});
            }

            var tapFn = taps[index++];
            // 调用插件，最后增加一个 next 回调
            tapFn(${params}, next);
          }

          next(null, ${firstArg});
        `;

          // 生成函数：最后一个参数永远是 callback
          return new Function("_x", ...this._args, "finalCallback", code);
        }

        callAsync(...args) {
          const finalCallback = args.pop(); // 最后一个是回调函数
          if (!this._callAsync) {
            this._callAsync = this.compile();
          }
          // 传入插件数组、用户参数、以及最终回调
          this._callAsync(this.taps, ...args, finalCallback);
        }
      }

      // --- 生产环境测试代码 ---

      const pipeline = new AsyncSeriesWaterfallHook(["source", "config"]);

      // 插件 A: 模拟异步 Babel 转换
      pipeline.tapAsync("Babel", (source, config, next) => {
        console.log(">>> 正在执行 Babel...");
        setTimeout(() => {
          const newSource = source + ` /* mode: ${config.mode} */`;
          next(null, newSource);
        }, 500);
      });

      // 插件 B: 模拟异步 Uglify 混淆
      pipeline.tapAsync("Uglify", (source, config, next) => {
        console.log(">>> 正在执行 Uglify...");
        setTimeout(() => {
          // 此时 source 已经是 Babel 处理过的了
          const newSource = source + " + Minified";
          next(null, newSource);
        }, 500);
      });

      // 执行
      console.time("cost");
      pipeline.callAsync("const a = 1;", { mode: "production" }, (err, result) => {
        if (err) console.error(err);
        console.log("最终产物:", result);
        console.timeEnd("cost"); // 预期耗时 1s 左右
      });
```

```js
// 引入我们之前实现的 Hook 类（这里简化展示）
class MiniBundler {
  constructor() {
    this.hooks = {
      // 1. 寻找入口：同步钩子
      beforeRun: new SyncHook(["compilerName"]),
      
      // 2. 转换代码：异步流水线（第一个参数是代码，第二个是配置）
      transform: new AsyncSeriesWaterfallHook(["source", "options"]),
      
      // 3. 输出文件：异步并行（多个任务同时跑，全完事才结束）
      emit: new AsyncParallelHook(["assets"])
    };
  }

  async run(entryCode) {
    console.log("=== 启动编译流程 ===\n");

    // 阶段 1: beforeRun
    this.hooks.beforeRun.call("MyBundler");

    // 阶段 2: transform (流水线加工)
    // 注意：这里需要支持异步
    const transformedCode = await this.hooks.transform.promise(entryCode, { minify: true });
    
    // 阶段 3: emit (并行分发)
    const assets = { "bundle.js": transformedCode };
    await this.hooks.emit.promise(assets);

    console.log("\n=== 编译成功 ===");
  }
}
```

```javascript
// ==================== 2. IoC 容器实现 ====================

class Container {
  constructor() {
    // 存储注册的类
    this.registrations = new Map();
    // 存储已创建的单例实例
    this.instances = new Map();
  }

  /**
   * 注册服务
   * @param {string} name - 服务名称
   * @param {Function} Class - 类定义
   * @param {string[]} deps - 依赖的服务名称列表
   */
  register(name, Class, deps = []) {
    this.registrations.set(name, { Class, deps });
  }

  /**
   * 解析并获取服务实例（核心方法）
   */
  resolve(name) {
    // 如果已经创建过，直接返回（单例）
    if (this.instances.has(name)) {
      console.log(`  ⚡ ${name} 已存在，复用实例`);
      return this.instances.get(name);
    }

    const registration = this.registrations.get(name);
    if (!registration) {
      throw new Error(`服务 ${name} 未注册`);
    }

    const { Class, deps } = registration;

    console.log(`🔧 正在创建 ${name}...`);

    // 递归解析所有依赖
    const resolvedDeps = deps.map(depName => this.resolve(depName));

    // 创建实例，注入依赖
    const instance = new Class(...resolvedDeps);

    // 缓存实例（单例）
    this.instances.set(name, instance);

    return instance;
  }
}


// ==================== 3. 使用容器 ====================

console.log('========== 注册服务 ==========\n');

const container = new Container();

// 注册所有服务，声明依赖关系
container.register('Database', MySQLDatabase, []);
container.register('UserService', UserService, ['Database']);
container.register('OrderService', OrderService, ['Database']);
container.register('PaymentService', PaymentService, ['UserService', 'OrderService']);

console.log('\n========== 首次获取 PaymentService ==========\n');

// 一行获取，容器自动处理整个依赖链
const payment = container.resolve('PaymentService');

console.log('\n========== 执行业务方法 ==========\n');

payment.pay(1);

console.log('========== 再次获取 UserService ==========\n');

// 再次获取，复用单例
const userService = container.resolve('UserService');
userService.getUser(2);
```
