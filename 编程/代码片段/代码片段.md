# 合并处理同一接口的并发请求
> 场景: 
  接口是支持 [id1,id2,id3]这种形式批量获取数据的。
  每个 webview 从 render 调用方法获取数据，如果 webview 多了可能会有大的并发请求，所以可能等 50 ms, 收集所有请求参数，然后发一个请求就好。
```javascript
export function createDebouncedBatcher<TId extends string, TResult, TItem = TResult>(
  fetcher: (ids: TId[]) => Promise<TResult>,
  options: DebouncedBatcherOptions<TId, TResult, TItem> = {}
): DebouncedBatcher<TId, TResult, TItem> {
  const defer = options.defer ?? 50;
  const pickById = options.pickById;

  let timer: ReturnType<typeof setTimeout> | null = null;
  let pendingIds = new Set<TId>();
  let waitersById = new Map<TId, Array<DebouncedBatcherDeferred<TItem>>>();
  let flushing = false;

  const cancelTimer = () => {
    if (timer) {
      clearTimeout(timer);
      timer = null;
    }
  };

  const scheduleFlush = (flushFn: () => void) => {
    if (timer) return;
    timer = setTimeout(() => {
      timer = null;
      flushFn();
    }, defer);
  };

  const flush = async (): Promise<void> => {
    if (flushing) return;
    if (pendingIds.size === 0) return;

    flushing = true;

    // 快照并清空（新 enqueue 进入下一批）
    const ids = Array.from(pendingIds);
    const localWaiters = waitersById;

    pendingIds = new Set<TId>();
    waitersById = new Map<TId, Array<DebouncedBatcherDeferred<TItem>>>();

    try {
      const result = await fetcher(ids);

      // 将同一批次 result 分发给每个 id 的所有 waiter
      for (const id of ids) {
        const waiters = localWaiters.get(id);
        if (!waiters || waiters.length === 0) continue;
        const picked = pickById ? pickById(result, id) : (result as unknown as TItem);
        waiters.forEach(({ resolve }) => resolve(picked));
      }
    } catch (error) {
      for (const id of ids) {
        const waiters = localWaiters.get(id);
        if (!waiters || waiters.length === 0) continue;
        waiters.forEach(({ reject }) => reject(error));
      }
    } finally {
      flushing = false;

      // 如果 flush 过程中有新 pending，尽快安排下一轮（不再等待 defer）
      if (pendingIds.size > 0 && !timer) {
        timer = setTimeout(() => {
          timer = null;
          void flush();
        }, 0);
      }
    }
  };

  const enqueue = (id: TId): Promise<TItem> => {
    pendingIds.add(id);

    const promise = new Promise<TItem>((resolve, reject) => {
      const list = waitersById.get(id) ?? [];
      list.push({ resolve, reject });
      waitersById.set(id, list);
    });

    scheduleFlush(() => {
      void flush();
    });

    return promise;
  };

  const flushNow = async (): Promise<void> => {
    cancelTimer();
    await flush();
  };

  return {
    enqueue,
    flushNow,
    cancelTimer,
    getPendingCount: () => pendingIds.size,
    isFlushing: () => flushing,
  };
}

```

# 惰性单例

**场景：** 你有一个很重的 SDK（比如地图、IM、支付），或者一个很贵的计算。你不想页面一加载就初始化，而想等到**第一次有人调用它**时再初始化，且只初始化一次。 **普通做法：** 全局变量 `let instance = null`，每次用的时候 `if (!instance) init()`。有点啰嗦。 **高级技巧：** **重写自身函数 (Function Overwriting)。**

```JavaScript
let getHeavySDK = async () => {
    console.log('🚧 第一次调用：正在初始化...');
    
    // 1. 执行初始化
    const sdk = await import('./heavy-sdk'); 
    const instance = sdk.createInstance();

    // 2. 【核心技巧】重写函数本身！
    // 下次再调用 getHeavySDK，直接执行下面这个函数，不再走上面的逻辑
    getHeavySDK = async () => {
        console.log('🚀 后续调用：直接返回实例');
        return instance;
    };

    return instance;
};

// --- 使用场景 ---
await getHeavySDK(); // 打印：🚧 第一次调用...
await getHeavySDK(); // 打印：🚀 后续调用... (连 if 判断都省了)
```

**原理：** 这是一个极其纯粹的性能优化，利用 JS 函数是一等公民的特性，在运行时动态修改函数的定义。

      class AsyncSeriesWaterfallHook {
        constructor(args = []) {
          this._args = args; // 示例: ["source", "config"]
          this.taps = [];
        }

        // 注册异步回调插件
        tapAsync(name, fn) {
          this.taps.push(fn);
        }

        compile() {
          const params = this._args.join(", "); // "source, config"
          const firstArg = this._args[0]; // "source"

          // 动态生成执行逻辑
          // 核心思想：手动构建一个递归函数 next(i, currentSource)
          let code = `
          var taps = _x;
          var index = 0;

          function next(err, newValue) {
            if (err) return finalCallback(err);
            if (newValue !== undefined) {
              ${firstArg} = newValue; // 更新流水线的值
            }

            if (index >= taps.length) {
              return finalCallback(null, ${firstArg});
            }

            var tapFn = taps[index++];
            // 调用插件，最后增加一个 next 回调
            tapFn(${params}, next);
          }

          next(null, ${firstArg});
        `;

          // 生成函数：最后一个参数永远是 callback
          return new Function("_x", ...this._args, "finalCallback", code);
        }

        callAsync(...args) {
          const finalCallback = args.pop(); // 最后一个是回调函数
          if (!this._callAsync) {
            this._callAsync = this.compile();
          }
          // 传入插件数组、用户参数、以及最终回调
          this._callAsync(this.taps, ...args, finalCallback);
        }
      }

      // --- 生产环境测试代码 ---

      const pipeline = new AsyncSeriesWaterfallHook(["source", "config"]);

      // 插件 A: 模拟异步 Babel 转换

// ==================== 2. IoC 容器实现 ====================

class Container {
  constructor() {
    // 存储注册的类
    this.registrations = new Map();
    // 存储已创建的单例实例
    this.instances = new Map();
  }

  /**
   * 注册服务
   * @param {string} name - 服务名称
   * @param {Function} Class - 类定义
   * @param {string[]} deps - 依赖的服务名称列表
   */
  register(name, Class, deps = []) {
    this.registrations.set(name, { Class, deps });
  }

  /**
   * 解析并获取服务实例（核心方法）
   */
  resolve(name) {
    // 如果已经创建过，直接返回（单例）
    if (this.instances.has(name)) {
      console.log(`  ⚡ ${name} 已存在，复用实例`);
      return this.instances.get(name);
    }

    const registration = this.registrations.get(name);
    if (!registration) {
      throw new Error(`服务 ${name} 未注册`);
    }

    const { Class, deps } = registration;

    console.log(`🔧 正在创建 ${name}...`);

    // 递归解析所有依赖
    const resolvedDeps = deps.map(depName => this.resolve(depName));

    // 创建实例，注入依赖
    const instance = new Class(...resolvedDeps);

    // 缓存实例（单例）
    this.instances.set(name, instance);

    return instance;
  }
}

// mapProps 是一个函数，接收底层的 Field 实例，返回需要覆盖/注入给 UI 组件的 Props
type MapPropsFn<P> = (field: Field) => Partial<P>;

function connect<P extends object>(
  TargetComponent: React.ComponentType<P>,
  mapProps?: MapPropsFn<P> // 允许外部传入自定义翻译规则
) {
  // 返回的高阶组件
  return (props: { field?: Field } & Omit<P, 'value' | 'onChange'>) => {
    const { field, ...restProps } = props;
    
    // 如果没有 field (独立使用组件)，直接原样渲染
    if (!field) return <TargetComponent {...(restProps as P)} />;

    // 1. 默认翻译规则：假设组件是标准的 value / onChange
    let mappedProps: Partial<P> = {
      value: field.state.value,
      onChange: (e: any) => field.onInput(e?.target && 'value' in e.target ? e.target.value : e),
      disabled: field.state.pattern === 'disabled',
      readOnly: field.state.pattern === 'readOnly',
    } as Partial<P>;

    // 2. ★ 核心：执行自定义翻译规则，覆盖默认规则 ★
    if (mapProps) {
      mappedProps = { ...mappedProps, ...mapProps(field) };
    }

    // 3. 将所有 props 合并注入给目标组件
    return <TargetComponent {...(restProps as any)} {...mappedProps} />;
  };
}

let currentListener: (() => void) | null = null;
function observable<T extends object>(target: T): T {
  const listeners = new Set<() => void>();
  return new Proxy(target, {
    get(obj, prop) {
      if (currentListener) listeners.add(currentListener);
      return obj[prop as keyof T];
    },
    set(obj, prop, value) {
      obj[prop as keyof T] = value;
      listeners.forEach(listener => listener());
      return true;
    }
  });
}
function useObserver(renderFn: () => React.ReactNode) {
  const [, forceUpdate] = useReducer((x) => x + 1, 0);
  currentListener = forceUpdate;
  const element = renderFn();
  currentListener = null;
  return element;
}

export class Form {
  public fields: Record<string, Field> = {};
  createField(props: IFieldProps): Field {
    if (!this.fields[props.name]) this.fields[props.name] = new Field(props);
    return this.fields[props.name];
  }
  query(name: string): Field | undefined { return this.fields[name]; }
}

// =========================================================================
// 第三部分：视图桥接层 (渲染核心)
// =========================================================================
const FormContext = createContext<Form | null>(null);
const SchemaComponentsContext = createContext<typeof componentsRegistry>(componentsRegistry);

export const FormProvider: React.FC<PropsWithChildren<{ form: Form; components: typeof componentsRegistry }>> = ({ form, components, children }) => (
  <FormContext.Provider value={form}>
    <SchemaComponentsContext.Provider value={components}>{children}</SchemaComponentsContext.Provider>
  </FormContext.Provider>
);

const FieldComponent: React.FC<IFieldProps> = (props) => {
  const form = useContext(FormContext)!;
  const registry = useContext(SchemaComponentsContext);
  const fieldInstance = form.createField(props);

  return useObserver(() => {
    const { state, component, decorator } = fieldInstance;
    if (state.display === 'none') return null;

    let content: React.ReactNode = null;
    if (component) {
      const [compName, compProps] = component;
      const RealComp = registry[compName]; 
      // 此时的 RealComp 已经是被 connect(含 mapProps) 包装过的组件了！
      if (RealComp) content = <RealComp {...(compProps as any)} field={fieldInstance} />;
    }

    if (decorator) {
       const [decName, decProps] = decorator;
       const RealDec = registry[decName];
       if (RealDec) content = <RealDec {...(decProps as any)} errors={state.errors}>{content}</RealDec>;
    }

    return <div style={{ display: state.display === 'hidden' ? 'none' : 'block' }}>{content}</div>;
  });
};

// =========================================================================
// 第四部分：Schema 解析引擎
// =========================================================================
export const SchemaField: React.FC<{ schema: ISchema; name?: string }> = ({ schema, name = "" }) => {
  const registry = useContext(SchemaComponentsContext);

  if ((schema.type === 'object' || schema.type === 'void') && schema.properties) {
    const getChildPath = (childKey: string) => schema.type === 'void' ? name : (name ? `${name}.${childKey}` : childKey); 
    const childrenNodes = Object.entries(schema.properties).map(([key, childSchema]) => (
      <SchemaField key={key} name={getChildPath(key)} schema={childSchema} />
    ));
    const ContainerComponent = schema['x-component'] ? registry[schema['x-component']] : Fragment;
    const containerProps = schema['x-component-props'] || {};
    return <ContainerComponent {...(containerProps as any)}>{childrenNodes}</ContainerComponent>;
  }

  return (
    <FieldComponent
      name={name}
      component={schema['x-component'] ? [schema['x-component'], schema['x-component-props']] as ComponentTuple : undefined}
      decorator={schema['x-decorator'] ? [schema['x-decorator'], schema['x-decorator-props']] as ComponentTuple : undefined}
    />
  );
};

const formSchema: ISchema = {
  type: 'object',
  properties: {
    systemConfig: {
      type: 'void',
      'x-component': 'Card',
      'x-component-props': { title: '系统配置 (验证 mapProps 适配能力)' },
      properties: {
        // 1. 标准 value 组件
        systemName: {
          type: 'string',
          'x-component': 'Input',
          'x-component-props': { placeholder: '请输入系统名称' },
          'x-decorator': 'FormItem',
          'x-decorator-props': { label: '系统名称', layout: 'horizontal' }
        },
        // 2. ★ 依赖 mapProps 翻译的特殊组件 ★
        enableLog: {
          type: 'boolean',
          'x-component': 'Switch', // 底层是 checkbox，需要 checked 属性
          'x-decorator': 'FormItem',
          'x-decorator-props': { label: '开启日志', layout: 'horizontal' }
        }
      }
    }
  }
};

const App = () => {
  const myForm = useMemo(() => {
    const form = new Form();
    
    // 初始化时给 boolean 类型赋个默认值
    setTimeout(() => {
      form.query('enableLog')?.onInput(true);
    }, 0);

    return form;
  }, []);

  return (
    <div style={{ padding: 40, maxWidth: 600, margin: '0 auto', fontFamily: 'sans-serif' }}>
      <FormProvider form={myForm} components={componentsRegistry}>
        <SchemaField schema={formSchema} />
      </FormProvider>
      
      <div style={{ marginTop: 20, padding: 15, background: '#e6f7ff', border: '1px solid #91d5ff', borderRadius: 4 }}>
        <h4 style={{ margin: '0 0 10px 0' }}>实时底层数据模型 (Model State):</h4>
        <pre style={{ margin: 0, fontSize: 13 }}>
          {/* 使用自定义的 useObserver 来让外部的 App 也能响应表单内部的数据变化，为了方便演示 */}
          {JSON.stringify({
             systemName: myForm.query('systemName')?.state.value,
             enableLog: myForm.query('enableLog')?.state.value // 无论 UI 怎么切，底层永远是纯粹的 Boolean
          }, null, 2)}
        </pre>
      </div>
    </div>
  );
};

export default App;
